- The forEach() method is used to iterate through an array and apply a specific function to each element
    - array.forEach(callback)
- Behind the scenes the forEach method will provide to a callback - element (current element), index (current index) and array (location of array - in this case 'numbers')
```
// Display all elements of an array
let numbers = [1,2,3,4,5];

function display(element){
    console.log(element)
}

numbers.forEach(display);

// Lets double each element of the array

let numbers = [1,2,4,5,6];

function double(element, index, array){
    array[index] = element * 2;
}

numbers.forEach(double);

```
