```
//JS file
const checkbox = document.getElementById("checkbox");
const visa = document.getElementById("visa");
const ms = document.getElementById("mastercard");
const paypal = document.getElementById("paypal");
const submit = document.getElementById("submit");
const subResult = document.getElementById("subResult");
const cardResult = document.getElementById("cardResult");

submit.onclick = function(){

    if(checkbox.checked){
        subResult.textContent = `You have subscribed!`; 
    }
    else{
        subResult.textContent = `You are not subscribed!`;
    }

    if(visa.checked){
        cardResults.textContent = `Visa`;
    }
    else if(ms.checked){
        cardResults.textContent = `MasterCard`;
    }
    else if(paypal.checked){
        cardResults.textContent = `Paypal`;
    }
    else{
        cardResults.textContent = `None`;
    }
}
```
