- These methods allow you to manipulate and work with strings
```
let username = "aj";

username.charAt(0) //get the char at 0 index

username.indexOf("a") // returns the index of the **first** occurrence of that letter
username.LastIndexOf("a") // returns the index of the **last** occurrence of that letter

username.length // returns length of the variable

username = username.trim(); // remove any whitespace in start of end

username.toUpperCase(); // make everything upper case
username.toLowerCase(); //make everything lowercase

username.repeat(3); // will repeat the string 3 times

let result = username.startsWith("a") // will return true or false depending on the string
let result = username.endsWith("a") // will return true or false depending on the string

let phone = "123-456-867787";
phone = phone.replaceAll("-",""); // replaces all occurrence of '-' with nothing

phone = phone.padStart(15, "0") // pad this string with 0's until its length is 15
phone = phone.padEnd(15, "0") // same for padding at end
