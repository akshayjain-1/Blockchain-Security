- callbacks: functions that are passed as an argument to another function
    - Used to handle Async operations:
        - Reading a file
        - Network requests
        - Interacting with db

- "Hey, when you are done, call this next"

```
function hello(){
    console.log("Hello");
}
function goodbye(){
    console.log("Goodbye");
}

hello();
goodbye();

// what is hello is taking too much time to process? JS maynot wait until hello is finished and may execute goodbye already.
// In case we want goodbye to be executed after hello, we update the function calls

function hello(callback){
    console.log("Hello");
    callback();
}

hello(goodbye); // do not add paranthesis in front of goodbye as that will call the function right away. Pass only the name

// You can pass other args along with callbacks as well.

function sum(callback, x, y){
    let result = x + y;
    callback(result);
}

function display(result){
    console.log(`${result}`);
}
sum(display, 5, 6);
